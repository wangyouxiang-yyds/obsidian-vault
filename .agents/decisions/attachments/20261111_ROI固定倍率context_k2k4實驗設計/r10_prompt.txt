"""Freeze / verify the ctx k2k4 v6 inputs.

  python freeze.py write --preflight <dir>   copy <dir>/preflight.json to W/preflight/ (refuse if present) and write
                                             FROZEN.sha256 over card, code, configs, every list file, cache manifest +
                                             provenance, preflight.json
  python freeze.py check                     launch-time integrity: preflight all gates passed, preflight code SHAs ==

import hashlib, json, shutil, sys
from pathlib import Path

HERE = Path(__file__).resolve().parent

CACHE = W / "cache_train_zstd"

        "build_cache.py", "make_configs.py", "launch.sh", "freeze.py", "val_fn_prereq.py", "run_val_fn_prereq.sh"]


def sha(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for b in iter(lambda: f.read(1 << 22), b""):
            h.update(b)
    return h.hexdigest()


def check():
    pf = json.loads((W / "preflight" / "preflight.json").read_text())
    if not pf["all_gates_passed"]:
        raise SystemExit("preflight gates not all passed")
    bad = [f for f, h in pf["code_sha256"].items() if sha(HERE / f) != h]
    if bad:
        raise SystemExit(f"code changed since preflight: {bad}")
    m = sha(CACHE / "cache_manifest.json")
    import yaml
    for k in (1, 2, 4):
        c = yaml.safe_load((W / "configs" / f"ctx_K{k}_S42.yaml").read_text())
        if c["ctx"]["cache_manifest_sha256"] != m or c["ctx"]["k"] != k:
            raise SystemExit(f"config K{k}: cache pin / k mismatch")
    if pf["G6_cache_equivalence"]["manifest_sha256"] != m:
        raise SystemExit("preflight G6 ran on another cache manifest")
    print("freeze check ok")


def write(pre):
    dst = W / "preflight" / "preflight.json"
    if dst.exists():
        raise SystemExit(f"{dst} exists")
    dst.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy2(Path(pre) / "preflight.json", dst)
    check()
    lm = json.loads((W / "lists" / "build_manifest.json").read_text())
    files = [HERE / f for f in CODE] + [W / "configs" / f"ctx_K{k}_S42.yaml" for k in (1, 2, 4)] + \
            [W / "lists" / f for f in sorted(lm["sha256"])] + [W / "lists" / "build_manifest.json"] + \
            [CACHE / "cache_manifest.json", CACHE / "index.json", CACHE / "cache_provenance.json", dst]
    for f in sorted(lm["sha256"]):
        if sha(W / "lists" / f) != lm["sha256"][f]:
            raise SystemExit(f"list {f} differs from build_manifest")
    lines = [f"{sha(p)}  {p if p.parent != HERE else p.name}" for p in files]
    (HERE / "FROZEN.sha256").write_text("\n".join(lines) + "\n")
    print(f"FROZEN.sha256: {len(lines)} entries")


if __name__ == "__main__":
    if sys.argv[1] == "check":
        check()
    elif sys.argv[1] == "write" and len(sys.argv) == 4 and sys.argv[2] == "--preflight":
        write(sys.argv[3])
    else:
        raise SystemExit(__doc__)

## 4. 模型（`ctx_common.CtxSeg`；decision v5）
- Prithvi-EO-2.0-300M（tap 23）+ DeepLabHead trunk（ASPP→3×3→BN→ReLU）共用，local 與 context 合併成 batch 32 一次 forward。
- f_c 以 affine_grid（scale 1/k）+ bilinear grid_sample（align_corners=False）對齊到 local footprint；K1 不取樣（identity）。
- 聚合：concat(512)→Conv3×3(512→256, no bias)→BN→ReLU→Conv1×1(256→2)；seeded Kaiming(fan_out)、BN 1/0、最後 bias 0。
- aux：P44 原 classifier（head[4]）共用，作用於 f_l 與完整 f_c。推論只用聚合輸出。
- Loss：`L = T(main) + 0.4·(T(aux_l) + T_soft(aux_c))/2`，Tversky α.3/β.7/smooth 1、batch 加總；soft 版在 hard target 上與 P44 TverskyLoss 相同（G3）。

## 5. Context 與 target（`ctx_common.make_pair`）
- 窗 (x,y) 的 context：從 (x−(k−1)·128, y−(k−1)·128) 讀 k·256 方窗；出影像範圍＝值 0（無效）、mask 255；可超出 ROI，不跨 VRT／split。
- 只讀模型 6 槽（B02 B03 B04 B8A B11 B12），移除未使用槽（0/4＝B01、B08，恆為 0）的讀取；模型 band_indices == [1,2,3,5,6,7] 於建模時 assert。（附帶發現：VRT 對 L8 B08 的 nearest 取樣值隨讀取窗起點而變；使用中的 6 槽另由 G6/G7 逐位元驗證與讀取窗無關。）
- area N/D（整數累加後一次除）：img = Σ(raw·v)/Σv/10000（Σv>0），否則 Σraw/k²/10000；D = Σv/k²；q = Σ(oil·v)/Σv；w = D ≥ .5，w=0 處 ignore。k=1 直接回傳 local（identity）。
- Label：pixel_mapping {0:油, 1/2/255:背景}；無效像素 ignore。

## 6. 訓練（`train_ctx.py`）

- I/O：TRAIN 讀無損 ZSTD 快取（`build_cache.py`；同解析度、uint16 原值、ROI 外擴 384；`cache_manifest.json` 記 lists SHA、建置碼 SHA、各景來源與 bbox、每檔 SHA，其 SHA 釘在 config；正式訓練啟動時全檔驗 SHA 並靜態確認 153,853 窗的 K4 footprint 全在快取內；嚴格模式，不得 fallback 到 VRT）。快取檔由先前 build_cache（npy 版，直接讀 VRT）產物以同一 `write()` 轉成 ZSTD；等價性以 G6 對 VRT 驗證。VAL／臺灣推論讀 VRT。16 workers、GDAL_CACHEMAX 128 MB。

- Probationary gate：前 200 步 ≥20 個含油 batch，其中 ≥18 個 context 半邊聚合梯度有限且 RMS>1e-12；未過即停、判 non-executed。

## 7. 判讀（`eval_ctx.py pilot`；決定順序固定）
1. collapse（best checkpoint 主 VAL：pooled recall ≤.01 或 pred/gt ≤.01）優先：K1 collapse → K2、K4 皆「不可判定（K1 collapse）」；Kk collapse → 該倍率「不可判定（treatment collapse）」。
2. Δmacro-3 = 各 run selection metric 之差（stride 256 無 TTA）；Δrecall_>20px = best.pt 以 stride 128＋orig/hflip/vflip 重跑 VAL，依 gate 1 方法（EDT 深度 >20 px）。
3. 「值得補 seed」：Δmacro-3 ≥ +0.05 且 Δrecall_>20px ≥ +0.10（不是追加訓練授權）。
4. 「pilot 負向」：Δmacro-3 ≤ −0.05。
5. Δmacro-3 ≥ +0.05 但 interior 未達：「不可判定—非預期效益訊號」。其餘：「不可判定」。
6. K1 新 VAL 表現型（描述）：top 元件佔 >20px GT >50% →「dominated by one scene/component」；否則 FN share_>20px ≥.30 且 recall_>20px ≤ recall_6–20px −.15 →「replicated」，否則「not replicated」。不改門檻／split、不重跑。
- 並列：pooled IoU、各 bin IoU/P/R/pred-gt、各景 IoU；TTA 版 macro-3 純描述。

## 8. 描述性分析（不參與分類）
- derangement（K2、K4；collapse 者仍算但標 interpretable=false）：主 VAL operator；每窗 context coverage 依 rank 分十分位；同 bin 內 donor 必須不同 duplicate_group_id（組依 group 排序後旋轉最大組數）；最大組 >半數的 bin 由低到高與異組候選較多的相鄰 bin 合併（平手取低、邊界向內）；seed 20261111；mapping 存 `eval/derangement_K{k}.json`。報 macro-3 下降。
- adoption：同 checkpoint，C_aligned ← f_l；A = ½[mean_oil|Δm| + mean_bg|Δm|]（m = z_oil − z_bg，有效像素）；A ≥ .01 記 adopted。
- 臺灣：pilot_result.json 寫出後才讀 GT；只報大林 2 張／其他 22 張 pooled IoU 與 Kk−K1；單 seed，描述性，不得選模。

## 9. 開跑前 gates（`preflight_ctx.py` → `preflight/preflight.json`，全過才可 launch）
G1 幾何（中心誤差 ≤.5 px；整數 N/D 與浮點參考 ≤1e-6）；G2 K1 identity（資料、特徵、aux ≤1e-6）；G3 soft Tversky 一致（FP32 五組 fixture：含油、全背景、混 ignore、全 ignore、batch16；loss ≤1e-6、grad 相對 ≤1e-5／近零時絕對 ≤1e-8）；G4 batch16 記憶體 ≤30.5 GiB；G5 K4 資料吞吐 ≥1.2× GPU 步速（worker 數＝正式 config 16）；G6 快取 manifest／全檔 SHA／釘選、全部 TRAIN 窗 K4 footprint 在快取內、每個 TRAIN 景各一正式窗（local/K2/K4）＋邊界與出界讀取與 VRT 逐位元相同；G7 跨 K local 一致（band_indices assert；TRAIN 快取路徑同窗同位置 K1/K2/K4 的 xl 與 mask 逐位元相同；VAL 每個 ROI 在三種讀取起點下 6 槽逐位元相同）；G8 隔離梯度（K2、K4 固定含油 batch：∂L_main/∂C_aligned、聚合 context 半邊、∂L_aux_ctx/∂f_c、∂L_aux_l/∂f_l 皆有限且 RMS>1e-12）。D1：各 split 以 seed 20261111 無放回抽 4096 窗，local/K2/K4 共用中心，報 raw-255 佔比（描述）。
正式 preflight（`W/preflight/preflight.json`，SHA-256 2d5f6678…c800；以凍結程式執行）：**G1–G8 全過**。G1 中心誤差 0、N/D 對參考 ≤6e-8；G2 差 0；G3 五組 fixture loss／grad 差 0；G4 峰值 12.68 GiB；G5 9.30 vs 6.99 batch/s（1.33×，workers 16）；G6 165 景、L8+S2、825 讀取 0 不符、TRAIN 窗出快取 0；G7 TRAIN 64 窗、VAL 33 ROI 0 不符；G8 四條梯度 RMS 最小 1.6e-6。
D1（揭露，不是 gate）：raw-255 佔比 TRAIN local .639／K2 .638／K4 .641；VAL .687／.687／.690（有效像素上相近），context 皆未超過 local 2 倍。raw 255 在 local 本身就佔六成以上（語義未文件化，P44 起即視為背景）。
快取血緣（`cache_provenance.json`）：現用檔案由早期 npy 版轉 ZSTD；以凍結版 `build_cache.py` 直接由 VRT 重建，330 檔 SHA 全同、index 內容相同。

## 10. 執行（`launch.sh`，需使用者另行授權）
K1 → K2 → K4 依序訓練（持 GPU 鎖）→ 三組 VAL TTA → pilot 分類 → K2/K4 反事實 → 三組臺灣預測（只讀影像）→ 臺灣評分 → `completion.json`（列出哪些產物存在；EXIT=0 不代表全部描述性分析都完成）。預估每 run 6–12 h（smoke：K4 5.7 it/s、每 VE 約 13–15 min）。
non-executed：讀取／對齊／identity 測試失敗、exposure gate 未過、OOM 後改 batch/loss、主終點不可還原。失敗不得改 seed、batch、算子、k、aux 或門檻。
