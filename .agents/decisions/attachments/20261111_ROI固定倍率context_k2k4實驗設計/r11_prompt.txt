def breakdown(preds, split, fb):
    rows, conf = [], []
    for scene, z in preds:
        gt = C.gt_crop(fb / split.loc[scene, "mask_path"], z["origin"], z["pred"].shape)
        c = C.confusion(z["pred"], gt, z["valid"], z["union"]); c.update(scene=scene, size_bin=split.loc[scene, "size_bin"])
        conf.append(c)
        m = z["valid"] & z["union"]
        g = (gt == 0) & m
        if not g.any():
            continue
        p = (z["pred"] == X.OIL) & m
        lab, _ = ndi.label(g, structure=np.ones((3, 3)))
        idx = np.nonzero(g)
        rows.append(pd.DataFrame(dict(scene=scene, tp=p[idx], comp=[f"{scene}#{v}" for v in lab[idx]],
                                      frac=ndi.uniform_filter(g.astype(np.float32), 256, mode="constant")[idx],
                                      depth=ndi.distance_transform_edt(g)[idx])))
    d = pd.concat(rows)
    d["dep_b"] = pd.cut(d.depth, DEP, labels=DEP_L, right=False)
    d["frac_b"] = pd.cut(d.frac, FRAC, labels=FRAC_L, right=False)
    tot = int((~d.tp).sum())
    tab = lambda col: d.groupby(col, observed=False).agg(gt_px=("tp", "size"), recall=("tp", "mean"),  # noqa: E731
                                                         fn=("tp", lambda s: int((~s).sum()))).assign(
        fn_share=lambda t: t.fn / max(tot, 1)).reset_index().astype({col: str}).to_dict("records")
    deep = d[d.depth > 20.5]
    mid = d[(d.depth > 5.5) & (d.depth <= 20.5)]
    top = deep.comp.value_counts()
    return dict(metrics=X.macro3(conf), recall_gt20px=float(deep.tp.mean()), gt_px_gt20px=len(deep),
                fn_share_gt20px=float((~deep.tp).sum() / max(tot, 1)), recall_6_20px=float(mid.tp.mean()),
                recall_ge80=float(d[d.frac >= .8].tp.mean()) if (d.frac >= .8).any() else None,
                top_component_share_gt20px=float(top.iloc[0] / len(deep)) if len(deep) else None,
                top_component_gt20px=top.index[0] if len(deep) else None,
                depth=tab("dep_b"), density=tab("frac_b"), per_scene=conf)

def window_coverage(split, cfg, k, scenes):
    """Per VAL window (main operator, stride 256): mean context coverage D (data only)."""
    w = pd.read_csv(L / "windows_val.tsv", sep="\t")
    w = w[w.scene.isin(scenes)].reset_index(drop=True)
    ds = T.CtxDataset([f"{s}_patch_x{x}_y{y}" for s, x, y in zip(w.scene, w.x, w.y)], split, cfg, k, SEED)
    cov = [ds.sample(i, 0, augment=False)[5] for i in range(len(ds))]
    w["coverage"] = cov
    w["group"] = split.loc[w.scene, "duplicate_group_id"].values
    return w, ds


def derange(w):
    """Coverage deciles (rank, ties by order); within bins donors from another duplicate group; infeasible bins
    (largest group > half) merged with the adjacent bin offering more other-group candidates (tie: lower index;
    edge bins merge inward), scanning low -> high, repeated until feasible."""
    rng = np.random.default_rng(PERM_SEED)
    w = w.copy()
    w["bin"] = (w.coverage.rank(method="first") - 1).mul(10).floordiv(len(w)).astype(int)
    bins = [list(w.index[w.bin == b]) for b in range(10)]

    def feasible(ix):
        return w.loc[ix, "group"].value_counts().iloc[0] * 2 <= len(ix)

    while True:
        bad = [i for i, b in enumerate(bins) if not feasible(b)]
        if not bad:
            break
        i = bad[0]
        if len(bins) == 1:
            raise SystemExit("derangement infeasible even with all windows in one bin")
        cand = [j for j in (i - 1, i + 1) if 0 <= j < len(bins)]
        own = set(w.loc[bins[i], "group"])
        score = {j: int((~w.loc[bins[j], "group"].isin(own)).sum()) for j in cand}
        j = max(cand, key=lambda c: (score[c], -c))
        lo, hi = min(i, j), max(i, j)
        bins = bins[:lo] + [bins[lo] + bins[hi]] + bins[hi + 1:]
    mapping = {}
    for b in bins:
        sub = w.loc[b]
        order = sub.sample(frac=1, random_state=rng.integers(2**31)).sort_values("group", kind="stable").index.tolist()
        n, shift = len(order), int(sub.group.value_counts().iloc[0])
        for a_, d_ in zip(order, order[shift:] + order[:shift]):
            if w.loc[a_, "group"] == w.loc[d_, "group"]:
                raise RuntimeError("same-group donor")
            mapping[a_] = d_
    return mapping, len(bins)


def score_rows(model, device, cfg, split, rois, perm_for=None, substitute=False, keep_margin=False):
    fb, vc = Path(cfg["paths"]["full_band"]), cfg["val"]
    rows, margins = [], {}
    for scene, g in rois.groupby("scene", sort=True):
        perm = perm_for(scene) if perm_for else None
        r = X.predict_scene(model, device, fb / split.loc[scene, "vrt_home"], None, g.to_dict("records"), cfg,
                            vc["stride"], vc["tta"], vc["amp"], vc["batch"], perm=perm, substitute_local=substitute)
        gt = C.gt_crop(fb / split.loc[scene, "mask_path"], r["origin"], r["pred"].shape)
        c = C.confusion(r["pred"], gt, r["valid"], r["union"]); c.update(scene=scene, size_bin=split.loc[scene, "size_bin"])
        rows.append(c)
        if keep_margin:
            m = r["valid"] & r["union"]
            margins[scene] = (r["margin"][m], (gt == 0)[m])
    return X.macro3(rows), margins


def cmd_counterfactual(a):
    k = a.k
    if k == 1:
        raise SystemExit("counterfactuals are defined for K2/K4")
    target = out_once(W / "eval" / f"counterfactual_K{k}.json")
    cfg, model, device, t = load(k)
    split = split_df()
    rois = val_rois()
    w, ds = window_coverage(split, cfg, k, set(rois.scene))
    mapping, nbins = derange(w)
    (W / "eval" / f"derangement_K{k}.json").write_text(json.dumps(
        [[*w.loc[a_, ["scene", "x", "y"]].tolist(), *w.loc[d_, ["scene", "x", "y"]].tolist()] for a_, d_ in mapping.items()],
        default=int))
    key = {(r.scene, r.x, r.y): i for i, r in w.iterrows()}

    def perm_for(scene):
        out = {}
        for i in w.index[w.scene == scene]:
            d = mapping[i]
            out[(int(w.loc[i, "x"]), int(w.loc[i, "y"]))] = ds.sample(d, 0, augment=False)[1]
        return out

    full, mf = score_rows(model, device, cfg, split, rois, keep_margin=True)
    der, _ = score_rows(model, device, cfg, split, rois, perm_for=perm_for)
    sub, ms = score_rows(model, device, cfg, split, rois, substitute=True, keep_margin=True)
    dm = np.concatenate([np.abs(mf[s][0] - ms[s][0]) for s in mf])
    oil = np.concatenate([mf[s][1] for s in mf])
    A = 0.5 * (float(dm[oil].mean()) + float(dm[~oil].mean()))
    coll = collapsed(rj(run_dir(k) / "val_best.json"))
    res = dict(k=k, checkpoint_sha256=t["best_sha256"], derangement_bins=nbins, n_windows=len(mapping),
               interpretable=not coll, collapse=coll,
               full=full, deranged=der, substituted=sub,
               derangement_drop_macro3=full["macro3"] - der["macro3"], A_identity=A, adopted=A >= 0.01,
               note="descriptive only (decision v6)", n_key=len(key))
    target.write_text(json.dumps(res, indent=1, default=float))
    print(json.dumps({kk: res[kk] for kk in ("derangement_drop_macro3", "A_identity", "adopted")}, indent=1))


# ----------------------------------------------------------------------------------------------- pilot
def collapsed(v):
    return v["all"]["recall"] <= 0.01 or v["all"]["pred_to_gt"] <= 0.01


def cmd_complete(_):
    """Completion manifest: which outputs exist (so EXIT=0 is not misread as 'all analyses present')."""
    want = [f"train_K{k}_S{SEED}/train_complete.json" for k in (1, 2, 4)] + \
           [f"eval/val_tta_K{k}.json" for k in (1, 2, 4)] + ["pilot_result.json"] + \
           [f"eval/counterfactual_K{k}.json" for k in (2, 4)] + \
           [f"predictions/CTX_K{k}/manifest.json" for k in (1, 2, 4)] + ["eval/taiwan_eval.json"]
    rep = {w_: (W / w_).exists() for w_ in want}
    (W / "completion.json").write_text(json.dumps(dict(present=rep, all_present=all(rep.values())), indent=1))
    print(json.dumps(rep, indent=1))


def cmd_pilot(_):
    target = out_once(W / "pilot_result.json")
    runs = {}
    for k in (1, 2, 4):
        t = rj(run_dir(k) / "train_complete.json")
        v = rj(run_dir(k) / "val_best.json")
        tta = rj(W / "eval" / f"val_tta_K{k}.json")
        if tta["checkpoint_sha256"] != t["best_sha256"]:
            raise SystemExit(f"K{k}: val_tta is not of best.pt")
        runs[k] = dict(macro3=t["best_val_macro3"], best_ve=t["best_virtual_epoch"], collapse=collapsed(v),
                       val_best=v, recall_gt20px_tta=tta["recall_gt20px"], tta_metrics=tta["metrics"],
                       recall_6_20px_tta=tta["recall_6_20px"], fn_share_gt20px_tta=tta["fn_share_gt20px"],
                       top_component_share_gt20px=tta["top_component_share_gt20px"])
    r1 = runs[1]
    ph = r1["fn_share_gt20px_tta"] >= 0.30 and r1["recall_gt20px_tta"] <= r1["recall_6_20px_tta"] - 0.15
    phenotype = ("dominated by one scene/component" if (r1["top_component_share_gt20px"] or 0) > 0.5
                 else "phenotype replicated" if ph else "phenotype not replicated")
    cls = {}
    for k in (2, 4):
        rk = runs[k]
        d_m, d_r = rk["macro3"] - r1["macro3"], rk["recall_gt20px_tta"] - r1["recall_gt20px_tta"]
        if r1["collapse"]:
            c = "不可判定（K1 collapse）"
        elif rk["collapse"]:
            c = "不可判定（treatment collapse）"
        elif d_m >= 0.05 and d_r >= 0.10:
            c = "值得補 seed"
        elif d_m <= -0.05:
            c = "pilot 負向"
        elif d_m >= 0.05:
            c = "不可判定—非預期效益訊號"
        else:
            c = "不可判定"
        cls[f"K{k}"] = dict(delta_macro3=d_m, delta_recall_gt20px=d_r, classification=c)
    res = dict(decision="v6", runs=runs, k1_phenotype=phenotype, classification=cls)
    target.write_text(json.dumps(res, indent=1, ensure_ascii=False, default=float))
    print(json.dumps(dict(k1_phenotype=phenotype, classification=cls), indent=1, ensure_ascii=False))

def cmd_taiwan(a):
    k = a.k
    final = W / "predictions" / f"CTX_K{k}"
    if final.exists():
        raise SystemExit(f"{final} exists")
    tmp = W / "predictions" / f".CTX_K{k}.tmp"
    shutil.rmtree(tmp, ignore_errors=True)
    tmp.mkdir(parents=True)
    cfg, model, device, t = load(k)
    tc = cfg["test"]
    rois = pd.read_csv(L / "rois_taiwan.csv")
    if SMOKE_TW:
        rois = rois[rois.scene.isin(sorted(rois.scene.unique())[:SMOKE_TW])]
    files = {}
    for i, scene in enumerate(sorted(rois.scene.unique())):
        g = rois[rois.scene == scene]
        r = X.predict_scene(model, device, g.vrt.iloc[0], None, g.to_dict("records"), cfg,
                            tc["stride"], tc["tta"], tc["amp"], tc["batch"])
        f = tmp / f"{i:02d}.npz"
        np.savez_compressed(f, scene=scene, origin=np.asarray(r["origin"]), pred=r["pred"], prob_oil=r["prob_oil"],
                            valid=r["valid"], union=r["union"])
        files[f.name] = dict(scene=scene, sha256=C.sha256(f))
        print(f"[taiwan K{k}] {i + 1}/{rois.scene.nunique()} {scene}", flush=True)
    (tmp / "manifest.json").write_text(json.dumps(dict(k=k, checkpoint_sha256=t["best_sha256"], operator=tc,
                                                       files=files), indent=1, ensure_ascii=False))
    os.replace(tmp, final)


def cmd_taiwan_eval(a):
    if not a.allow_target_gt:
        raise SystemExit("needs --allow-target-gt")
    if not (W / "pilot_result.json").exists():
        raise SystemExit("Taiwan is read only after the VAL pilot classification is frozen")
    target = out_once(W / "eval" / "taiwan_eval.json")
    reg = {s["scene_id"]: s for s in rj(REGISTRY)["all_scenes"]}
    res = {}
    for k in (1, 2, 4):
        d = W / "predictions" / f"CTX_K{k}"
        man = rj(d / "manifest.json")
        rows = []
        for fname, info in man["files"].items():
            if C.sha256(d / fname) != info["sha256"]:
                raise SystemExit(f"{d}/{fname} changed")
            z = np.load(d / fname)
            scene = str(z["scene"])
            gt = C.gt_crop(reg[scene]["mask"], tuple(int(v) for v in z["origin"]), z["pred"].shape)
            c = C.confusion(z["pred"], gt, z["valid"], z["union"]); c["subset"] = "dalin" if DALIN in scene else "other"
            rows.append(c)
        if sum(r["subset"] == "dalin" for r in rows) != 2 or len(rows) != 24:
            raise SystemExit("subsets are not 2 + 22")
        res[f"K{k}"] = {s: C.pooled([r for r in rows if r["subset"] == s]) for s in ("dalin", "other")}
    res["delta"] = {f"K{k}": {s: res[f"K{k}"][s]["iou"] - res["K1"][s]["iou"] for s in ("dalin", "other")} for k in (2, 4)}
    res["note"] = "descriptive only; single seed; never used for selection (decision v6)"
    target.write_text(json.dumps(res, indent=1, default=float))
    print(json.dumps(res["delta"], indent=1))

    if a.smoke_work:
