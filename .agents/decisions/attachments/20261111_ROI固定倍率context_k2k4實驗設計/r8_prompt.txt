# ----------------------------------------------------------------------------------------------- model
class CtxSeg(nn.Module):
    def __init__(self, base, k: int, init_seed: int):
        super().__init__()
        self.k = int(k)
        self.encoder = base.encoder
        self.register_buffer("band_indices", base.band_indices.clone(), persistent=False)
        self.feature_index = base.feature_index
        self.trunk = nn.Sequential(*list(base.head.children())[:4])
        self.aux = list(base.head.children())[4]
        cpu = torch.Generator().manual_seed(int(init_seed))
        conv3 = nn.Conv2d(512, 256, 3, padding=1, bias=False)
        conv1 = nn.Conv2d(256, 2, 1)
        with torch.no_grad():
            for m in (conv3, conv1):
                w = torch.empty_like(m.weight)
                nn.init.kaiming_normal_(w, mode="fan_out", nonlinearity="relu", generator=cpu)
                m.weight.copy_(w)
            conv1.bias.zero_()
        bn = nn.BatchNorm2d(256)
        nn.init.ones_(bn.weight)
        nn.init.zeros_(bn.bias)
        self.agg = nn.Sequential(conv3, bn, nn.ReLU(inplace=True), conv1)

    def features(self, x):
        x = x.index_select(1, self.band_indices).unsqueeze(2)
        f = self.encoder.forward_features(x)[self.feature_index]
        return self.encoder.prepare_features_for_image_model([f])[0]

    def align(self, fc):
        if self.k == 1:
            return fc
        B = fc.shape[0]
        theta = torch.zeros(B, 2, 3, device=fc.device, dtype=fc.dtype)
        theta[:, 0, 0] = theta[:, 1, 1] = 1.0 / self.k
        grid = F.affine_grid(theta, list(fc.shape), align_corners=False)
        return F.grid_sample(fc, grid, mode="bilinear", padding_mode="zeros", align_corners=False)

    def forward(self, xl, xc, substitute_local=False):
        B, hw = xl.shape[0], xl.shape[-2:]
        f = self.trunk(self.features(torch.cat([xl, xc], 0)))
        fl, fc = f[:B], f[B:]
        ca = fl if substitute_local else self.align(fc)
        up = lambda t: F.interpolate(t, size=hw, mode="bilinear", align_corners=False)  # noqa: E731
        return dict(out=up(self.agg(torch.cat([fl, ca], 1))), aux_l=up(self.aux(fl)), aux_c=up(self.aux(fc)),
                    fl=fl, ca=ca)

    def head_parameters(self):
        return list(self.trunk.parameters()) + list(self.aux.parameters()) + list(self.agg.parameters())


def build(cfg, k, init_seed, checkpoint=None, expected_sha=None):
    base, device = C.build_model(cfg)
    m = CtxSeg(base, k, init_seed).to(device)
    if checkpoint is not None:
        if expected_sha is not None and C.sha256(checkpoint) != expected_sha:
            raise RuntimeError(f"checkpoint SHA mismatch: {checkpoint}")
        st = torch.load(checkpoint, map_location="cpu", weights_only=False)
        m.load_state_dict(st["model_state"] if "model_state" in st else st, strict=True)
    return m, device


# ----------------------------------------------------------------------------------------------- loss
class SoftTversky(nn.Module):
    """P44 TverskyLoss (alpha .3, beta .7, smooth 1, batch-summed) with a soft target q in [0,1] and weight vm."""

    def __init__(self, alpha=0.3, beta=0.7, smooth=1.0):
        super().__init__()
        self.alpha, self.beta, self.smooth = alpha, beta, smooth

    def forward(self, logits, q, vm):
        p = torch.softmax(logits, dim=1)[:, OIL]
        TP = (p * q * vm).sum()
        FP = (p * (1 - q) * vm).sum()
        FN = ((1 - p) * q * vm).sum()
        return 1 - (TP + self.smooth) / (TP + self.alpha * FP + self.beta * FN + self.smooth)


def hard_to_soft(mask):
    """int mask {0 oil, 1 bg, 255 ignore} -> (q, vm) float."""
    return (mask == OIL).float(), (mask != IGN).float()


def total_loss(o, mask, cq, cvm, tv=SoftTversky()):
    q, vm = hard_to_soft(mask)
    main = tv(o["out"].float(), q, vm)
    al = tv(o["aux_l"].float(), q, vm)
    ac = tv(o["aux_c"].float(), cq, cvm)
    return main + AUX_W * (al + ac) / 2, dict(main=main, aux_l=al, aux_c=ac)

def area(a, k):
    """(..., k*n, k*n) -> (..., n, n) block mean (float32). Reference operator (preflight G1)."""
    a = a.astype(np.float32, copy=False)
    if k == 1:
        return a
    *lead, H, W = a.shape
    return a.reshape(*lead, H // k, k, W // k, k).mean(axis=(-3, -1), dtype=np.float32)


def _lut(pixel_mapping):
    lut = np.full(256, IGN, np.int64)
    for kk, v in pixel_mapping.items():
        lut[int(kk)] = int(v)
    return lut


def make_pair(raw6, rm, k, pixel_mapping):
    """raw6 (6, kP, kP) uint16 VRT values of the model slots, rm (kP, kP) uint8 raw mask (context window, local at
    its centre) -> local clean (8,P,P), local label {0,1,255}, context clean (8,P,P), coverage D, soft target q, w.

    Identical to the clean() / invalid_from_raw() / area N-D definitions: VRT values are uint16 integers, so
    clean = x/10000 and invalid = any model slot == 0. Context (k > 1) is computed in integers then divided once:
    img = sum(raw*v)/sum(v)/10000 (D > 0) else sum(raw)/k^2/10000; D = sum(v)/k^2; q = sum(oil*v)/sum(v);
    w = D >= .5. k = 1 returns the local tensors themselves (identity). Unused slots 0/4 stay 0."""
    o = (k - 1) * P // 2
    valid = (raw6 > 0).all(axis=0)
    lab = _lut(pixel_mapping)[rm]
    if (lab == IGN).any():
        raise RuntimeError("unmapped raw label")
    li = np.zeros((8, P, P), np.float32)
    li[SLOTS] = raw6[:, o:o + P, o:o + P].astype(np.float32) / 10000.0
    lv = valid[o:o + P, o:o + P]
    ll = lab[o:o + P, o:o + P].copy()
    if k == 1:
        q = ((ll == OIL) & lv).astype(np.float32)
        w = lv.astype(np.float32)
        ll[~lv] = IGN
        return li, ll, li.copy(), w.copy(), q, w
    ll[~lv] = IGN
    blk = lambda a, ax: a.reshape(*a.shape[:-2], P, k, P, k).sum(axis=ax, dtype=np.uint32)  # noqa: E731
    Dn = blk(valid, (1, 3)).astype(np.float64)
    N = blk(raw6 * valid, (2, 4)).astype(np.float64)
    img6 = N / np.maximum(Dn, 1)
    if (Dn == 0).any():   # fully invalid cells: plain block mean (only computed when needed)
        img6 = np.where(Dn > 0, img6, blk(raw6, (2, 4)).astype(np.float64) / (k * k))
    img6 = img6 / 10000.0
    ci = np.zeros((8, P, P), np.float32)
    ci[SLOTS] = img6
    oil = blk((lab == OIL) & valid, (1, 3)).astype(np.float64)
    w = (2 * Dn >= k * k)
    q = np.where(Dn > 0, oil / np.maximum(Dn, 1), 0.0) * w
    return li, ll, ci, (Dn / (k * k)).astype(np.float32), q.astype(np.float32), w.astype(np.float32)

def read_raw(src, msrc, x, y, side):

    rm = np.full((side, side), 255, np.uint8)
    if x1 > x0 and y1 > y0:
        win = Window(x0, y0, x1 - x0, y1 - y0)
        raw[:, y0 - y:y1 - y, x0 - x:x1 - x] = _read6(src, win)
        if msrc is not None:
            rm[y0 - y:y1 - y, x0 - x:x1 - x] = msrc.read(1, window=win)
    return raw, rm


class _CachedBand:
    """rasterio-like reader over the lossless ZSTD cache (build_cache.py); VRT fallback outside its bbox."""

    def __init__(self, ds, box, fallback, is_mask):
        self.ds, self.box, self.fallback, self.is_mask = ds, box, fallback, is_mask
        self.height, self.width = box["H"], box["W"]
        self._fb = None

    def _inside(self, win):
        x0, y0 = int(win.col_off), int(win.row_off)
        b = self.box
        if b["x0"] <= x0 and x0 + int(win.width) <= b["x1"] and b["y0"] <= y0 and y0 + int(win.height) <= b["y1"]:
            return Window(x0 - b["x0"], y0 - b["y0"], int(win.width), int(win.height))
        return None

    def _fallback(self):
        if self._fb is None:
            self._fb = rasterio.open(self.fallback)
        return self._fb

    def read6(self, win):
        w = self._inside(win)
        return self.ds.read(window=w) if w is not None else _read6(self._fallback(), win)

    def read(self, band, window):
        w = self._inside(window)
        return self.ds.read(band, window=w) if w is not None else self._fallback().read(band, window=window)

def ctx_origin(x, y, k):
    o = (k - 1) * P // 2
    return x - o, y - o


# ----------------------------------------------------------------------------------------------- augmentation
def aug_params(rng: np.random.Generator):
    """P44 PARITY augmentation, re-implemented so local and context share every parameter."""
    p = {}
    if rng.random() < 0.5:   # RandomResizedCrop(256, scale (.9,1), ratio (3/4,4/3))
        for _ in range(10):
            s = rng.uniform(0.9, 1.0) * P * P
            r = np.exp(rng.uniform(np.log(3 / 4), np.log(4 / 3)))
            w, h = int(round(np.sqrt(s * r))), int(round(np.sqrt(s / r)))
            if 0 < w <= P and 0 < h <= P:
                p["rrc"] = (int(rng.integers(0, P - w + 1)), int(rng.integers(0, P - h + 1)), w, h)
                break
    p["hflip"] = rng.random() < 0.5
    p["vflip"] = rng.random() < 0.5
    p["rot90"] = int(rng.integers(0, 4)) if rng.random() < 0.8 else 0
    if rng.random() < 0.4:
        p["noise"] = (float(rng.uniform(0.005, 0.02)), int(rng.integers(0, 2**31 - 1)))
    if rng.random() < 0.3:
        p["bc"] = (1.0 + float(rng.uniform(-0.1, 0.1)), float(rng.uniform(-0.1, 0.1)))
    return p


def _crop(a, cx, cy, w, h, interp):
    """Resample the (w x h) box centred at (cx, cy) (edge coords; box = [cx-w/2, cx+w/2)) to P x P. a: (H,W) or (H,W,C)."""
    sx, sy = w / P, h / P
    M = np.array([[sx, 0, cx - w / 2 + 0.5 * sx - 0.5], [0, sy, cy - h / 2 + 0.5 * sy - 0.5]], np.float64)
    return cv2.warpAffine(a, M, (P, P), flags=interp | cv2.WARP_INVERSE_MAP, borderMode=cv2.BORDER_REFLECT_101)


def geom(a, p, k, interp, delta=None):
    """Apply RRC (context: same w,h, centre shifted by delta/k), flips, rot90 to (H,W[,C])."""
    if "rrc" in p:
        x0, y0, w, h = p["rrc"]
        dx, dy = x0 + w / 2 - P / 2, y0 + h / 2 - P / 2
        a = _crop(a, P / 2 + dx / k, P / 2 + dy / k, w, h, interp)
    if p["hflip"]:
        a = a[:, ::-1]
    if p["vflip"]:
        a = a[::-1]
    if p["rot90"]:
        a = np.rot90(a, p["rot90"])
    return np.ascontiguousarray(a)


def photometric(img_hwc, p):
    if "noise" in p:
        std, s = p["noise"]
        img_hwc = img_hwc + np.random.default_rng(s).normal(0, std, img_hwc.shape).astype(np.float32)
    if "bc" in p:
        a, b = p["bc"]
        img_hwc = img_hwc * a + b
    return img_hwc.astype(np.float32)


# ----------------------------------------------------------------------------------------------- scoring
def macro3(rows):
    """rows: per-scene confusion dicts with key size_bin -> dict(all=pooled, small/medium/large=pooled, macro3)."""
    out = {"all": C.pooled(rows)}
    for b in BINS:
        out[b] = C.pooled([r for r in rows if r["size_bin"] == b])
    out["macro3"] = float(np.mean([out[b]["iou"] for b in BINS]))
    return out


# ----------------------------------------------------------------------------------------------- inference
@torch.no_grad()
def forward_pair(model, xl, xc, tta, amp, substitute_local=False):
    def f(a, b):
        return C._finite(model(a, b, substitute_local)["out"].float(), "ctx")
    with torch.autocast("cuda", enabled=amp):
        out = f(xl, xc)
        if tta:
            out = out + torch.flip(f(torch.flip(xl, [3]), torch.flip(xc, [3])), [3])
            out = out + torch.flip(f(torch.flip(xl, [2]), torch.flip(xc, [2])), [2])
            out = out / 3.0
    return C._finite(out, "fused")


class _Null:
    def __enter__(self):
        return None

    def __exit__(self, *a):
        return False


@torch.no_grad()
def predict_scene(model, device, vrt, mask_path, rois, cfg, stride, tta, amp, batch=32, perm=None,
                  substitute_local=False):
    """Context-aware logit mosaic over a scene's ROI union (P44 predict_scene + context per window).
    mask_path None -> labels never opened (context targets are not used at inference).
    perm: optional {(x, y): context tensor} replacement (derangement). Returns P44 dict + coverage per window."""
    ds = cfg["dataset"]
    k = model.k
    ux0, uy0 = min(r["rx0"] for r in rois), min(r["ry0"] for r in rois)
    ux1, uy1 = max(r["rx1"] for r in rois), max(r["ry1"] for r in rois)
    h, w = uy1 - uy0, ux1 - ux0
    acc = np.zeros((2, h, w), np.float32)
    cnt = np.zeros((h, w), np.uint16)
    valid = np.zeros((h, w), bool)
    union = np.zeros((h, w), bool)
    wins = R.roi_windows(rois, stride)
    done, cov = set(), {}
    model.eval()
    o = (k - 1) * P // 2
    with rasterio.open(vrt) as src, (rasterio.open(mask_path) if mask_path else _Null()) as msrc:
        C.check_scene_grid(src, int(rois[0]["H"]), int(rois[0]["W"]), rois)
        for r in rois:
            rx0, ry0, rx1, ry1 = r["rx0"], r["ry0"], r["rx1"], r["ry1"]
            ex0, ey0 = rx0 - o, ry0 - o
            raw, rm = read_raw(src, msrc, ex0, ey0, rx1 - rx0 + 2 * o)
            sl = (slice(ry0 - uy0, ry1 - uy0), slice(rx0 - ux0, rx1 - ux0))
            valid[sl] = (raw[:, o:o + ry1 - ry0, o:o + rx1 - rx0] > 0).all(axis=0)
            union[sl] = True
            todo = [(x, y) for x, y in wins if (x, y) not in done
                    and rx0 <= x and x + P <= rx1 and ry0 <= y and y + P <= ry1]
            done.update(todo)
            s = k * P
            for i in range(0, len(todo), batch):
                chunk = todo[i:i + batch]
                xl, xc = [], []
                for x, y in chunk:
                    cx, cy = x - o - ex0, y - o - ey0
                    li, _, ci, D, _, _ = make_pair(raw[:, cy:cy + s, cx:cx + s], rm[cy:cy + s, cx:cx + s], k,
                                                   ds["pixel_mapping"])
                    cov[(x, y)] = float(D.mean())
                    xl.append(C.normalize(li, ds["mean"], ds["std"]))
                    xc.append(C.normalize(ci, ds["mean"], ds["std"]))
                xl = torch.from_numpy(np.stack(xl)).to(device)
                xc = torch.from_numpy(np.stack(xc)).to(device)
                if perm is not None:
                    xc = torch.stack([perm[(x, y)].to(device) if (x, y) in perm else xc[j]
                                      for j, (x, y) in enumerate(chunk)])
                logits = forward_pair(model, xl, xc, tta, amp, substitute_local).cpu().numpy()
                for (x, y), lg in zip(chunk, logits):
                    acc[:, y - uy0:y - uy0 + P, x - ux0:x - ux0 + P] += lg
                    cnt[y - uy0:y - uy0 + P, x - ux0:x - ux0 + P] += 1
    if len(done) != len(wins):
        raise RuntimeError(f"{len(wins) - len(done)} windows not assigned to any ROI")
    c = cnt > 0
    if not np.array_equal(c, union):
        raise RuntimeError("window coverage != ROI union")
    mean_logit = acc / np.maximum(cnt, 1)
    pred = np.where(c, np.where(mean_logit[0] >= mean_logit[1], OIL, BG), 255).astype(np.uint8)
    e = np.exp(mean_logit - mean_logit.max(axis=0, keepdims=True))
    prob = (e[0] / e.sum(axis=0)).astype(np.float16)
    prob[~c] = 0
    margin = (mean_logit[0] - mean_logit[1]).astype(np.float32)
    return dict(origin=(ux0, uy0), pred=pred, prob_oil=prob, valid=valid, union=union, coverage=cov, margin=margin)

class PosSampler(VirtualEpochSampler):
    """P44 VirtualEpochSampler, yielding (index, global stream position)."""

def parse(name):
    stem, xy = name.rsplit("_patch_x", 1)
    x, y = xy.split("_y")
    return stem, int(x), int(y)


class CtxDataset(Dataset):
    def __init__(self, files, split, cfg, k, seed, cache=None):
        self.items = [parse(f) for f in files]
        self.cache = cache   # lossless same-resolution read cache (build_cache.py); None -> VRT
        self.split, self.k, self.seed = split, int(k), int(seed)
        self.ds = cfg["dataset"]
        self.fb = Path(cfg["paths"]["full_band"])
        self._h = {}

    def __len__(self):
        return len(self.items)

    def _open(self, path):
        if path not in self._h:
            self._h[path] = rasterio.open(path)
        return self._h[path]

    def _pair(self, scene):
        vrt, mask = self.fb / self.split.loc[scene, "vrt_home"], self.fb / self.split.loc[scene, "mask_path"]
        if self.cache and scene not in self._h:
            self._h[scene] = X.cached_pair(self.cache, scene, vrt, mask)
        if self.cache and self._h[scene] is not None:
            return self._h[scene]
        return self._open(vrt), self._open(mask)

    def sample(self, idx, pos, augment=True):
        scene, x, y = self.items[idx]
        k, P = self.k, X.P
        src, msrc = self._pair(scene)
        cx, cy = X.ctx_origin(x, y, k)
        raw, rm = X.read_raw(src, msrc, cx, cy, k * P)
        li, ll, ci, cov, cq, cw = X.make_pair(raw, rm, k, self.ds["pixel_mapping"])
        p = X.aug_params(np.random.default_rng([self.seed, int(pos)])) if augment else \
            dict(hflip=False, vflip=False, rot90=0)
        li = X.photometric(X.geom(li.transpose(1, 2, 0), p, 1, cv2.INTER_LINEAR), p)
        ci = X.photometric(X.geom(ci.transpose(1, 2, 0), p, k, cv2.INTER_LINEAR), p)
        ll = X.geom(ll.astype(np.uint8), p, 1, cv2.INTER_NEAREST).astype(np.int64)
        cq = X.geom(cq, p, k, cv2.INTER_NEAREST)
        cw = X.geom(cw, p, k, cv2.INTER_NEAREST)
        norm = lambda a: C.normalize(a.transpose(2, 0, 1), self.ds["mean"], self.ds["std"]).astype(np.float32)  # noqa: E731
        return (torch.from_numpy(norm(li)), torch.from_numpy(norm(ci)), torch.from_numpy(ll),
                torch.from_numpy(cq), torch.from_numpy(cw), float(cov.mean()))

    def __getitem__(self, item):
        idx, pos = item
        return self.sample(idx, pos)

def collate(b):
    xl, xc, m, cq, cw, cov = zip(*b)
    return torch.stack(xl), torch.stack(xc), torch.stack(m), torch.stack(cq), torch.stack(cw), torch.tensor(cov)


def param_groups(model, lr0, mult):
    enc = list(model.encoder.parameters())
    head = model.head_parameters()
    allp = [p for p in model.parameters() if p.requires_grad]
    if len(enc) + len(head) != len(allp) or len({id(p) for p in enc + head}) != len(allp):
        raise RuntimeError("parameter groups do not partition the model")
    return [{"params": enc, "lr": lr0 * mult}, {"params": head, "lr": lr0}]

def validate(model, device, cfg, val_rois, split, stride, tta, amp, batch):
    fb = Path(cfg["paths"]["full_band"])
    rows = []
    for scene, g in val_rois.groupby("scene", sort=True):
        r = X.predict_scene(model, device, fb / split.loc[scene, "vrt_home"], None,
                            g.to_dict("records"), cfg, stride, tta, amp, batch)
        gt = C.gt_crop(fb / split.loc[scene, "mask_path"], r["origin"], r["pred"].shape)
        c = C.confusion(r["pred"], gt, r["valid"], r["union"])
        c.update(scene=scene, size_bin=split.loc[scene, "size_bin"])
        rows.append(c)
    return X.macro3(rows), rows

    lr0 = float(tc["lr0"])

    max_ve = int(tc["max_virtual_epochs"]) if args.mode == "train" else args.max_ve
    max_steps = spv * max_ve
    sched = torch.optim.lr_scheduler.CosineAnnealingLR(opt, T_max=max_steps, eta_min=lr0 * float(tc["eta_min_ratio"]))
    amp = bool(tc["use_amp"])
    scaler = torch.cuda.amp.GradScaler(enabled=amp)
    patience = int(tc["patience"]) if args.mode == "train" else args.patience
    vc = cfg["val"]
    resolved = dict(config=json.loads(json.dumps(cfg)), mode=args.mode, k=k, n_train_windows=len(files),
                    steps_per_virtual_epoch=spv, max_virtual_epochs=max_ve, max_steps=max_steps, patience=patience,
                    val_scenes=int(val_rois.scene.nunique()), lists=str(lists),
                    lists_manifest_sha256=C.sha256(lists / "build_manifest.json"),
                    code_sha256={f: C.sha256(HERE / f) for f in TRAIN_CODE},
                    p44_code_sha256={f: C.sha256(X.P44 / f) for f in ("p44_common.py", "p44_roi.py", "train_p44.py")})
    (out / "resolved_config.json").write_text(json.dumps(resolved, indent=1, ensure_ascii=False))
    board = Board(args.mode == "train", out, resolved, f"CTX_K{k}_S{seed}")
    hist_f = open(out / "history.csv", "w", newline="")
    hist, perm_log, expo = None, [], dict(oil=0, ok=0, rms=[])
    best, best_ve, bad, step, ok = -1.0, -1, 0, 0, False
    ctx_w = model.agg[0].weight
    try:
        for ve in range(max_ve):
            sampler.set_epoch(ve)
            idx = sampler.indices(ve)
            perm_log.append(dict(ve=ve, first=idx[:3], last=idx[-3:],
                                 sha=hashlib.sha256(np.asarray(idx, np.int64).tobytes()).hexdigest()))
            model.train()
            t0 = time.time()
            agg = dict(loss=0.0, main=0.0, aux_l=0.0, aux_c=0.0, n=0, oil_batches=0, cov=0.0,
                       nonfinite_grad=0, scale_drops=0, gctx=[])
            for xl, xc, m, cq, cw, cov in loader:
                xl, xc, m = xl.to(device, non_blocking=True), xc.to(device, non_blocking=True), m.to(device)
                cq, cw = cq.to(device), cw.to(device)
                with torch.autocast("cuda", enabled=amp):
                    o = model(xl, xc)
                    loss, parts = X.total_loss(o, m, cq, cw)
                if not torch.isfinite(loss):
                    raise RuntimeError(f"non-finite loss at step {step}")
                scaler.scale(loss).backward()
                scaler.unscale_(opt)
                g = ctx_w.grad[:, 256:]
                g_rms = float(g.float().pow(2).mean().sqrt()) if g is not None else float("nan")
                gn = float(torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=float(tc["grad_clip"])))
                s_before = scaler.get_scale()
                scaler.step(opt)
                scaler.update()
                opt.zero_grad(set_to_none=True)
                sched.step()
                step += 1
                has_oil = bool((m == 0).any())
                if step <= EXPOSURE_STEPS and has_oil:
                    expo["oil"] += 1
                    expo["ok"] += int(np.isfinite(g_rms) and g_rms > 1e-12)
                    expo["rms"].append(g_rms)
                if step == EXPOSURE_STEPS:
                    (out / "exposure_gate.json").write_text(json.dumps(dict(
                        oil_batches=expo["oil"], ok_batches=expo["ok"],
                        median_rms=float(np.median(expo["rms"])) if expo["rms"] else None,
                        passed=expo["oil"] >= EXPOSURE_MIN_OIL and expo["ok"] >= EXPOSURE_MIN_OK), indent=1))
                    if args.mode == "train" and not (expo["oil"] >= EXPOSURE_MIN_OIL and expo["ok"] >= EXPOSURE_MIN_OK):
                        raise RuntimeError(f"exposure gate failed: {expo['oil']} oil batches, {expo['ok']} ok")
                agg["loss"] += float(loss)
                for kk in ("main", "aux_l", "aux_c"):
                    agg[kk] += float(parts[kk])
                agg["n"] += 1
                agg["oil_batches"] += int(has_oil)
                agg["cov"] += float(cov.mean())
                agg["nonfinite_grad"] += int(not np.isfinite(gn))
                agg["scale_drops"] += int(scaler.get_scale() < s_before)
                agg["gctx"].append(g_rms)
                if step % 200 == 0:
                    print(f"[train] step {step} loss {agg['loss']/agg['n']:.4f} main {agg['main']/agg['n']:.4f} "
                          f"aux_c {agg['aux_c']/agg['n']:.4f} oil_batch_frac {agg['oil_batches']/agg['n']:.3f} "
                          f"{agg['n']/(time.time()-t0):.2f} it/s mem {torch.cuda.max_memory_reserved()/2**30:.1f}GiB",

def breakdown(preds, split, fb):
    rows, conf = [], []
    for scene, z in preds:
        gt = C.gt_crop(fb / split.loc[scene, "mask_path"], z["origin"], z["pred"].shape)
        c = C.confusion(z["pred"], gt, z["valid"], z["union"]); c.update(scene=scene, size_bin=split.loc[scene, "size_bin"])
        conf.append(c)
        m = z["valid"] & z["union"]
        g = (gt == 0) & m
        if not g.any():
            continue
        p = (z["pred"] == X.OIL) & m
        lab, _ = ndi.label(g, structure=np.ones((3, 3)))
        idx = np.nonzero(g)
        rows.append(pd.DataFrame(dict(scene=scene, tp=p[idx], comp=[f"{scene}#{v}" for v in lab[idx]],
                                      frac=ndi.uniform_filter(g.astype(np.float32), 256, mode="constant")[idx],
                                      depth=ndi.distance_transform_edt(g)[idx])))
    d = pd.concat(rows)
    d["dep_b"] = pd.cut(d.depth, DEP, labels=DEP_L, right=False)
    d["frac_b"] = pd.cut(d.frac, FRAC, labels=FRAC_L, right=False)
    tot = int((~d.tp).sum())
    tab = lambda col: d.groupby(col, observed=False).agg(gt_px=("tp", "size"), recall=("tp", "mean"),  # noqa: E731
                                                         fn=("tp", lambda s: int((~s).sum()))).assign(
        fn_share=lambda t: t.fn / max(tot, 1)).reset_index().astype({col: str}).to_dict("records")
    deep = d[d.depth > 20.5]
    mid = d[(d.depth > 5.5) & (d.depth <= 20.5)]
    top = deep.comp.value_counts()
    return dict(metrics=X.macro3(conf), recall_gt20px=float(deep.tp.mean()), gt_px_gt20px=len(deep),
                fn_share_gt20px=float((~deep.tp).sum() / max(tot, 1)), recall_6_20px=float(mid.tp.mean()),
                recall_ge80=float(d[d.frac >= .8].tp.mean()) if (d.frac >= .8).any() else None,
                top_component_share_gt20px=float(top.iloc[0] / len(deep)) if len(deep) else None,
                top_component_gt20px=top.index[0] if len(deep) else None,
                depth=tab("dep_b"), density=tab("frac_b"), per_scene=conf)

def window_coverage(split, cfg, k, scenes):
    """Per VAL window (main operator, stride 256): mean context coverage D (data only)."""
    w = pd.read_csv(L / "windows_val.tsv", sep="\t")
    w = w[w.scene.isin(scenes)].reset_index(drop=True)
    ds = T.CtxDataset([f"{s}_patch_x{x}_y{y}" for s, x, y in zip(w.scene, w.x, w.y)], split, cfg, k, SEED)
    cov = [ds.sample(i, 0, augment=False)[5] for i in range(len(ds))]
    w["coverage"] = cov
    w["group"] = split.loc[w.scene, "duplicate_group_id"].values
    return w, ds


def derange(w):
    """Coverage deciles (rank, ties by order); within bins donors from another duplicate group; infeasible bins
    (largest group > half) merged with the adjacent bin offering more other-group candidates (tie: lower index;
    edge bins merge inward), scanning low -> high, repeated until feasible."""
    rng = np.random.default_rng(PERM_SEED)
    w = w.copy()
    w["bin"] = (w.coverage.rank(method="first") - 1).mul(10).floordiv(len(w)).astype(int)
    bins = [list(w.index[w.bin == b]) for b in range(10)]

    def feasible(ix):
        return w.loc[ix, "group"].value_counts().iloc[0] * 2 <= len(ix)

    while True:
        bad = [i for i, b in enumerate(bins) if not feasible(b)]
        if not bad:
            break
        i = bad[0]
        if len(bins) == 1:
            raise SystemExit("derangement infeasible even with all windows in one bin")
        cand = [j for j in (i - 1, i + 1) if 0 <= j < len(bins)]
        own = set(w.loc[bins[i], "group"])
        score = {j: int((~w.loc[bins[j], "group"].isin(own)).sum()) for j in cand}
        j = max(cand, key=lambda c: (score[c], -c))
        lo, hi = min(i, j), max(i, j)
        bins = bins[:lo] + [bins[lo] + bins[hi]] + bins[hi + 1:]
    mapping = {}
    for b in bins:
        sub = w.loc[b]
        order = sub.sample(frac=1, random_state=rng.integers(2**31)).sort_values("group", kind="stable").index.tolist()
        n, shift = len(order), int(sub.group.value_counts().iloc[0])
        for a_, d_ in zip(order, order[shift:] + order[:shift]):
            if w.loc[a_, "group"] == w.loc[d_, "group"]:
                raise RuntimeError("same-group donor")
            mapping[a_] = d_
    return mapping, len(bins)


def score_rows(model, device, cfg, split, rois, perm_for=None, substitute=False, keep_margin=False):
    fb, vc = Path(cfg["paths"]["full_band"]), cfg["val"]
    rows, margins = [], {}

        perm = perm_for(scene) if perm_for else None
        r = X.predict_scene(model, device, fb / split.loc[scene, "vrt_home"], None, g.to_dict("records"), cfg,
                            vc["stride"], vc["tta"], vc["amp"], vc["batch"], perm=perm, substitute_local=substitute)
        gt = C.gt_crop(fb / split.loc[scene, "mask_path"], r["origin"], r["pred"].shape)

        rows.append(c)

def cmd_counterfactual(a):

    dm = np.concatenate([np.abs(mf[s][0] - ms[s][0]) for s in mf])

    A = 0.5 * (float(dm[oil].mean()) + float(dm[~oil].mean()))
    res = dict(k=k, checkpoint_sha256=t["best_sha256"], derangement_bins=nbins, n_windows=len(mapping),

               note="descriptive only (decision v6)", n_key=len(key))

def cmd_taiwan_eval(a):

def sha256(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for b in iter(lambda: f.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()


def dp_split(pool: pd.DataFrame) -> set[str]:
    vec = pool.groupby("duplicate_group_id").size_bin.agg(lambda s: tuple(int((s == b).sum()) for b in BINS))
    ids = sorted(vec.index)
    order = [ids[i] for i in np.random.default_rng(SPLIT_SEED).permutation(len(ids))]
    reach = {(0, 0, 0): None}                       # state -> (prev_state, group_id)
    for gid in order:
        v = vec[gid]
        for s in sorted(reach):                     # snapshot, lexicographic
            n = tuple(a + b for a, b in zip(s, v))
            if max(n) <= PER_BIN and n not in reach:
                reach[n] = (s, gid)
    target = (PER_BIN,) * 3
    if target not in reach:
        raise SystemExit("target (10,10,10) unreachable under group constraint; fail-closed")
    chosen, s = set(), target
    while reach[s] is not None:
        s, gid = reach[s]
        chosen.add(gid)
    return chosen


def footprints(scenes, vrt_of):

def box84(crs, tf, x0, y0, x1, y1):
    a, b = tf * (x0, y0), tf * (x1, y1)
    return transform_bounds(crs, "EPSG:4326", min(a[0], b[0]), min(a[1], b[1]), max(a[0], b[0]), max(a[1], b[1]))


def hit(a, b):
    return not (a[2] <= b[0] or b[2] <= a[0] or a[3] <= b[1] or b[3] <= a[1])

"""Lossless same-resolution read cache for TRAIN (I/O gate G5 remedy; values bit-identical to the VRT read).

Source TIFs are LZW, 2-row strips at full width, so any window read decodes whole image rows. Per TRAIN scene we store
the union bbox of its ROIs expanded by MARGIN = (4-1)*128 (largest context), clipped to the image: the six model
slots (B02 B03 B04 B8A B11 B12; slots 0/4 are zeroed for every read, see ctx_common.make_pair) as uint16 and the
mask as uint8, each a 256-tiled ZSTD(level 1, predictor 2) GeoTIFF (lossless). VRT = 8 nearest SimpleSources of
uint16 TIFs -> integer float32; checked here: every value integral and in [0, 65535]. ctx_common.cached_pair falls
back to the VRT for any window not fully inside the cached bbox. Bitwise equivalence: preflight G6.
"""
from __future__ import annotations

import argparse
import json
import os
import sys
from multiprocessing import Pool
from pathlib import Path

import numpy as np
import pandas as pd
import rasterio
from rasterio.windows import Window

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE))
MARGIN = 384
FB = Path("/home/alanyh/oil_dataset/new/full_band")


MODEL_SLOTS = [1, 2, 3, 5, 6, 7]


def write(path, a):
    with rasterio.open(path, "w", driver="GTiff", width=a.shape[2], height=a.shape[1], count=a.shape[0],
                       dtype=a.dtype, tiled=True, blockxsize=256, blockysize=256, compress="ZSTD", zstd_level=1,
                       predictor=2, interleave="pixel") as d:
        d.write(a)


def job(args):
    scene, vrt, mask, rois, out = args
    with rasterio.open(vrt) as src, rasterio.open(mask) as msrc:
        H, W = src.height, src.width
        x0 = max(min(r[0] for r in rois) - MARGIN, 0); y0 = max(min(r[1] for r in rois) - MARGIN, 0)
        x1 = min(max(r[2] for r in rois) + MARGIN, W); y1 = min(max(r[3] for r in rois) + MARGIN, H)
        win = Window(x0, y0, x1 - x0, y1 - y0)
        a = src.read(window=win)
        if not (np.isfinite(a).all() and (a == np.round(a)).all() and a.min() >= 0 and a.max() <= 65535):
            raise RuntimeError(f"{scene}: VRT values not uint16-exact")
        m = msrc.read(1, window=win)
    write(out / f"{scene}.img.tif", a[MODEL_SLOTS].astype(np.uint16))
    write(out / f"{scene}.mask.tif", m[None].astype(np.uint8))
    return scene, dict(x0=int(x0), y0=int(y0), x1=int(x1), y1=int(y1), H=int(H), W=int(W), count=int(a.shape[0]))


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--lists", required=True)
    ap.add_argument("--out", required=True)
    ap.add_argument("--workers", type=int, default=6)
    a = ap.parse_args()
    out = Path(a.out)
    if (out / "index.json").exists():
        raise SystemExit("cache exists")
    out.mkdir(parents=True, exist_ok=True)
    L = Path(a.lists)
    split = pd.read_csv(L / "split_scenes.csv").set_index("scene_id")
    rois = pd.read_csv(L / "rois_train.csv")
    jobs = [(s, str(FB / split.loc[s, "vrt_home"]), str(FB / split.loc[s, "mask_path"]),
             list(zip(g.rx0, g.ry0, g.rx1, g.ry1)), out) for s, g in rois.groupby("scene")]
    with Pool(a.workers) as p:
        idx = dict(p.imap_unordered(job, jobs))
    (out / "index.json").write_text(json.dumps(dict(margin=MARGIN, scenes=idx), indent=1, ensure_ascii=False))
    print(len(idx), "scenes cached")


if __name__ == "__main__":
    main()

"""ctx k2k4 v6 preflight gates (fail-closed). Writes <out>/preflight.json.

G1 geometry   : (a) feature-level alignment of a coordinate ramp, centre error <= 0.5 source px (K2, K4);
                (b) real data: centre (P/k)^2 block of the context image == area(local, k) (<= 1e-6).
G2 K1 identity: dataset (with augmentation) xc == xl, ctx target == local target; model fl == fc, ca == fl,
                aux_c == aux_l (all <= 1e-6).
G3 soft Tversky == P44 TverskyLoss on hard targets: |loss| <= 1e-6, grad rel <= 1e-5.
G4 memory     : K4 batch 16 training steps, peak reserved <= 30.5 GiB.

def g1_geometry(split, files):
    res = {}
    for k in (2, 4):
        # (a) context feature cell j centre (source px, rel. to context origin) = (j+.5)*16k; local cell i centre
        # rel. to context origin = o + (i+.5)*16. Align a ramp feature map and compare.
        o = (k - 1) * X.P // 2
        ramp = ((torch.arange(16, dtype=torch.float64) + 0.5) * 16 * k)
        fc = ramp.view(1, 1, 1, 16).expand(1, 1, 16, 16).contiguous()
        m = X.CtxSeg.__new__(X.CtxSeg)
        torch.nn.Module.__init__(m)
        m.k = k
        got = m.align(fc)[0, 0, 8]
        want = o + (torch.arange(16, dtype=torch.float64) + 0.5) * 16
        err_feat = float((got - want).abs().max())
        # (b) real data: centre (P/k)^2 block of the context == float32 area() of the local clean image, and
        #     the full context == float reference N/D over the full-res clean window (<= 1e-6)
        ds = T.CtxDataset(files[::5000][:8], split, cfg_k(k), k, 42)
        errs, ref_errs = [], []
        for i in range(len(ds)):
            scene, x, y = ds.items[i]
            src, msrc = ds._pair(scene)
            cx, cy = X.ctx_origin(x, y, k)
            raw, rm = X.read_raw(src, msrc, cx, cy, k * X.P)
            li, ll, ci, D, q, w = X.make_pair(raw, rm, k, ds.ds["pixel_mapping"])
            img = np.zeros((8, k * X.P, k * X.P), np.float32)
            img[X.SLOTS] = raw.astype(np.float32) / 10000.0
            v = (raw > 0).all(0).astype(np.float32)
            Dr = X.area(v, k)
            ref = np.where(Dr[None] > 0, X.area(img * v, k) / np.where(Dr > 0, Dr, 1)[None], X.area(img, k))
            ref_errs.append(float(np.abs(ci - ref).max()))
            if not (raw[:, o:o + X.P, o:o + X.P] > 0).all():
                continue
            c0 = o // k
            errs.append(float(np.abs(ci[:, c0:c0 + X.P // k, c0:c0 + X.P // k] - X.area(li, k)).max()))
        res[f"K{k}"] = dict(feature_centre_err_px=err_feat, block_err=max(errs) if errs else None, n_block=len(errs),
                            ref_err=max(ref_errs))
    passed = all(v["feature_centre_err_px"] <= 0.5 and v["block_err"] is not None and v["block_err"] <= 1e-6
                 and v["ref_err"] <= 1e-6 for v in res.values())
    return dict(passed=passed, **res)


def g2_k1_identity(split, files, model, device):
    ds = T.CtxDataset(files[::997][:64], split, cfg_k(1), 1, 42)
    md = 0.0
    batch = []
    for i in range(len(ds)):
        xl, xc, m, cq, cw, _ = ds.sample(i, 1000 + i)
        q, vm = X.hard_to_soft(m)
        md = max(md, float((xl - xc).abs().max()), float((q - cq).abs().max()), float((vm - cw).abs().max()))
        batch.append((xl, xc))
    xl = torch.stack([b[0] for b in batch[:16]]).to(device)
    xc = torch.stack([b[1] for b in batch[:16]]).to(device)
    model.eval()
    with torch.no_grad():
        o = model(xl, xc)
    fd = float((o["fl"] - o["ca"]).abs().max())
    ad = float((o["aux_l"] - o["aux_c"]).abs().max())
    return dict(passed=md <= 1e-6 and fd <= 1e-6 and ad <= 1e-6, data_max_diff=md, feature_max_diff=fd, aux_max_diff=ad)


def g3_tversky(device):
    g = torch.Generator().manual_seed(0)
    logits = torch.randn(4, 2, 64, 64, generator=g, dtype=torch.float64)
    mask = torch.randint(0, 2, (4, 64, 64), generator=g)
    mask[:, :5] = 255
    a, b = logits.clone().requires_grad_(), logits.clone().requires_grad_()
    l1 = TverskyLoss(alpha=0.3, beta=0.7, oil_class=0, ignore_index=255)(a, mask)
    q, vm = X.hard_to_soft(mask)
    l2 = X.SoftTversky()(b, q.double(), vm.double())
    l1.backward(); l2.backward()
    rel = float((a.grad - b.grad).norm() / a.grad.norm())
    return dict(passed=abs(float(l1 - l2)) <= 1e-6 and rel <= 1e-5, loss_diff=abs(float(l1 - l2)), grad_rel=rel)


def g6_cache_equivalence(split, files, n=300):
    """Cached reader == VRT reader, bitwise, for local/K2/K4 windows incl. image-edge and out-of-image ones."""
    cfg = cfg_k(4)
    cache = Path(cfg["ctx"]["cache"])
    ds_c = T.CtxDataset(files, split, cfg, 4, 42, cache)
    ds_v = T.CtxDataset(files, split, cfg, 4, 42, None)
    rng = np.random.default_rng(7)
    picks = rng.choice(len(files), n, replace=False)
    worst, checked, hits = 0, 0, 0
    for i in picks:
        scene, x, y = ds_c.items[i]
        sc, mc = ds_c._pair(scene)
        sv, mv = ds_v._pair(scene)
        for side, (xx, yy) in ((256, (x, y)), (512, X.ctx_origin(x, y, 2)), (1024, X.ctx_origin(x, y, 4)),
                               (1024, (-200, y)), (1024, (sc.width - 600, sc.height - 300))):
            a = X.read_raw(sc, mc, xx, yy, side)
            b = X.read_raw(sv, mv, xx, yy, side)
            same = all(np.array_equal(p, q) for p, q in zip(a, b))
            worst += int(not same)
            checked += 1
        b = sc.box
        hits += int(b["x0"] <= X.ctx_origin(x, y, 4)[0] and X.ctx_origin(x, y, 4)[0] + 1024 <= b["x1"])
    for i in picks[:16]:   # whole training sample, K4, with augmentation
        p, q = ds_c.sample(int(i), 5), ds_v.sample(int(i), 5)
        same = all(torch.equal(u, v) if torch.is_tensor(u) else u == v for u, v in zip(p, q))
        worst += int(not same)
        checked += 1
    rep = dict(passed=worst == 0, mismatches=worst, checked=checked, k4_windows_inside_cache=hits, n_windows=n)
    (cache / "equivalence.json").write_text(json.dumps(rep, indent=1))
    return rep


def g4_g5(split, files, steps, io_batches, workers):

    scaler = torch.cuda.amp.GradScaler()
    ds = T.CtxDataset(files, split, cfg, 4, 42, cfg["ctx"]["cache"])

                    pin_memory=True, persistent_workers=False, worker_init_fn=T.worker_init)
    it = iter(dl)
    pre = [next(it) for _ in range(4)]
    torch.cuda.reset_peak_memory_stats()
    model.train()
    times = []
    for s in range(steps):
        xl, xc, m, cq, cw, _ = pre[s % len(pre)]
        torch.cuda.synchronize(); t = time.time()
        with torch.autocast("cuda"):
            o = model(xl.to(device), xc.to(device))
            loss, _ = X.total_loss(o, m.to(device), cq.to(device), cw.to(device))
        scaler.scale(loss).backward()
        scaler.unscale_(opt)
        torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        scaler.step(opt); scaler.update(); opt.zero_grad(set_to_none=True)
        torch.cuda.synchronize(); times.append(time.time() - t)
    peak = torch.cuda.max_memory_reserved() / 2**30
    gpu_rate = 1.0 / float(np.median(times[5:]))
    del model, opt
    torch.cuda.empty_cache()
    t = time.time()
    n = 0
    for _ in range(io_batches):
        next(it); n += 1
    io_rate = n / (time.time() - t)
    return (dict(passed=peak <= 30.5, peak_reserved_gib=peak),
            dict(passed=io_rate >= 1.2 * gpu_rate, data_batches_per_s=io_rate, gpu_steps_per_s=gpu_rate,
                 ratio=io_rate / gpu_rate, workers=workers, batches=n))


def d1_raw255(split, files, vfiles, n=300):
    out = {}
    for name, fl in (("train", files), ("val", vfiles)):
        sel = fl[:: max(1, len(fl) // n)][:n]
        for k in (2, 4):
            ds = T.CtxDataset(sel, split, cfg_k(k), k, 42)
            loc = ctx = loc_v = ctx_v = 0.0
            for i in range(len(ds)):
                scene, x, y = ds.items[i]
                src, msrc = ds._pair(scene)
                cx, cy = X.ctx_origin(x, y, k)
                raw, rm = X.read_raw(src, msrc, cx, cy, k * X.P)
                valid = (raw > 0).all(0)
                o = (k - 1) * X.P // 2
                lr, lvv = rm[o:o + X.P, o:o + X.P], valid[o:o + X.P, o:o + X.P]
                loc += (lr == 255).mean(); ctx += (rm == 255).mean()
                loc_v += ((lr == 255) & lvv).sum() / max(lvv.sum(), 1); ctx_v += ((rm == 255) & valid).sum() / max(valid.sum(), 1)
            m = len(ds)
            r = dict(local_255=loc / m, context_255=ctx / m, local_255_on_valid=loc_v / m, context_255_on_valid=ctx_v / m)
            r["context_gt_2x_local_on_valid"] = bool(r["context_255_on_valid"] > 2 * r["local_255_on_valid"])
            out[f"{name}_K{k}"] = r
    return out

#!/bin/bash
# ctx k2k4 v6 pilot: K1 -> K2 -> K4 (S42) training, then VAL TTA breakdowns, counterfactuals (K2/K4),
# frozen pilot classification, image-only Taiwan predictions, and last the Taiwan scoring (Dalin 2 / other 22).
# Holds the shared GPU lock for the whole sequence. Needs a clean tree and FROZEN.sha256 match.
set -euo pipefail
here=$(cd "$(dirname "$0")" && pwd)
repo=$(cd "$here/../.." && pwd)
W=/home/alanyh/oil_dataset/new/_ctx_k2k4_20261111
lock=/home/alanyh/oil_dataset/new/.gpu_main_runner.lock
log=$W/launch.log
git_=(git -c safe.directory="$repo" -C "$repo")
cd "$here"
if ! "${git_[@]}" diff --quiet -- analysis/ctx_k2k4_20261111 || \
   "${git_[@]}" status --porcelain -- analysis/ctx_k2k4_20261111 | grep -q '^?? '; then
    echo 'ctx_k2k4 has uncommitted or untracked files; commit before launch' >&2; exit 2
fi
sha256sum -c --quiet FROZEN.sha256 || { echo 'code differs from FROZEN.sha256' >&2; exit 2; }
python -c "import json,sys; r=json.load(open('$W/preflight/preflight.json')); sys.exit(0 if r['all_gates_passed'] else 1)" \
    || { echo 'preflight gates not all passed' >&2; exit 2; }

exec 9>"$lock" || { echo "cannot open GPU lock $lock" >&2; exit 3; }
printf '[CTX] waiting for GPU lock at %s\n' "$(date -Is)" >> "$log"
flock 9 || { echo 'flock failed' >&2; exit 3; }
printf '[CTX] start=%s commit=%s pid=%s\n' "$(date -Is)" "$("${git_[@]}" rev-parse HEAD)" "$$" >> "$log"

        --out "$W/train_K${k}_S42" > "$W/train_K${k}.log" 2>&1

step "pilot" python eval_ctx.py pilot >> "$W/eval.log" 2>&1
for k in 2 4; do step "counterfactual K$k" python eval_ctx.py counterfactual --k $k >> "$W/eval.log" 2>&1 || printf "[CTX] counterfactual K%s FAILED (descriptive; continuing)\n" "$k" >> "$log"; done
for k in 1 2 4; do step "taiwan K$k" python eval_ctx.py taiwan --k $k >> "$W/eval.log" 2>&1; done
step "taiwan_eval" python eval_ctx.py taiwan_eval --allow-target-gt >> "$W/eval.log" 2>&1
printf '[CTX] EXIT=0 %s\n' "$(date -Is)" >> "$log"
